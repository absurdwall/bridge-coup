import datetime
import json
import pathlib
import sys

repo = pathlib.Path('/Users/tingranwang/.codex/worktrees/issue46-integration/bridge-coup')
sys.path.insert(0, str(repo / 'evals/stage1'))
import acceptance

output = repo / 'evals/stage1/evidence/issue46-completion/development-v4'
native = pathlib.Path('/Users/tingranwang/.nvm/versions/node/v24.12.0/lib/node_modules/@openai/codex/node_modules/@openai/codex-darwin-arm64/vendor/aarch64-apple-darwin/bin/codex')
candidate = acceptance.freeze(output, repo / '.build/debug/bridge-play', native)
schedule = json.loads((output / 'plan.json').read_text())
ids = ['short-3-r1', 'short-5-r1', 'short-6-r1']
attempts = [next(a for a in schedule if a['id'] == name) for name in ids]
plan = {
    'purpose': 'Bounded development check, not the 22-attempt final acceptance matrix.',
    'candidate_source': candidate['source_commit'],
    'change': 'Align root output schema, existing field enumeration and examples to plan, selectedCard, publicReason. No content or strategy guidance changes.',
    'hypothesis': 'An instruction/schema process-order inconsistency is corrected; improvement in model quality is unproven.',
    'attempts': attempts,
    'policy': 'One fresh independent run per selected scenario; run all three sequentially; no playing retries or answer repair. Preserve all results. A clear failure does not qualify this candidate for final acceptance. If all three are clean, a separate complete 22-attempt matrix is still required.',
    'unchanged': ['Luna Medium player', 'Sol Medium grader', 'corpus', 'rubric', 'hands and facts', 'plan semantics', 'history and all_turns', 'empty tool surface', 'parser and legal-card enum'],
}
acceptance.write(output / 'DEVELOPMENT-PLAN.json', plan)
rows = []
for attempt in attempts:
    acceptance.preflight(candidate)
    result = acceptance.run_attempt(candidate, attempt, output)
    rows.append(result)
    acceptance.write(output / 'development-summary.json', {'candidate_source': candidate['source_commit'], 'attempts': rows, 'status': 'development-only-no-go' if any(a['outcome'] == 'fail' for a in rows) else 'development-only-pending-review'})
    print(f"{attempt['id']}: {result['outcome']}, legal={result['legal_execution']}, quality={result['quality']}, counts={result['quality_counts']}", flush=True)
acceptance.write(output / 'execution-finished.json', {'utc': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'attempts': len(rows), 'purpose': 'development-only'})
